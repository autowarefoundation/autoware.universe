cmake_minimum_required(VERSION 3.14)
project(autoware_safety_planner)

find_package(autoware_cmake REQUIRED)
autoware_package()

generate_parameter_library(safety_planner_parameters
  param/safety_planner_parameter_struct.yaml
)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)

ament_auto_add_library(${PROJECT_NAME}_common STATIC
  src/utils/boundary_simplifier.cpp
  src/utils/frenet_utils.cpp
  src/utils/route_tracking.cpp
  src/utils/trajectory_postprocess.cpp
  src/utils/turn_indicator_decider.cpp
  src/utils/velocity_optimizer.cpp
)
target_link_libraries(${PROJECT_NAME}_common
  safety_planner_parameters
)
target_compile_options(${PROJECT_NAME}_common PRIVATE -Wno-deprecated-declarations)

ament_auto_add_library(${PROJECT_NAME}_constraint_plugins SHARED
  src/constraint_generator/external_velocity_limit.cpp
  src/constraint_generator/lane_following_drivable_area.cpp
  src/constraint_generator/lanelet_speed_limit.cpp
  src/constraint_generator/obstacle_stop.cpp
  src/constraint_generator/vehicle_kinematics.cpp
)
target_link_libraries(${PROJECT_NAME}_constraint_plugins
  safety_planner_parameters
  ${PROJECT_NAME}_common
)
target_compile_options(${PROJECT_NAME}_constraint_plugins PRIVATE -Wno-deprecated-declarations)


# The trajectory planner plugins; compiling the constraints is an implementation detail of this .so
ament_auto_add_library(${PROJECT_NAME}_trajectory_planner_plugins SHARED
  src/trajectory_planner/frenet_sampling_based_planner/compiled_constraints_utils.cpp
  src/trajectory_planner/frenet_sampling_based_planner/constraints_compiler.cpp
  src/trajectory_planner/frenet_sampling_based_planner/frenet_sampling_based_planner.cpp
  src/trajectory_planner/reference_path_following_planner/reference_path_following_planner.cpp
)
target_link_libraries(${PROJECT_NAME}_trajectory_planner_plugins
  safety_planner_parameters
  ${PROJECT_NAME}_common
)
target_compile_options(${PROJECT_NAME}_trajectory_planner_plugins PRIVATE -Wno-deprecated-declarations)

# The MPPI planner needs the CUDA library of autoware_mppi_optimizer, which that package only
# exports when it was built with CUDA; the rest of the package does not depend on it


pluginlib_export_plugin_description_file(${PROJECT_NAME} plugins.xml)

ament_auto_add_library(${PROJECT_NAME}_lib SHARED
  src/safety_planner.cpp
  src/safety_planner_node.cpp
  src/utils/reference_path_smoother.cpp
)
target_link_libraries(${PROJECT_NAME}_lib
  safety_planner_parameters
  ${PROJECT_NAME}_common
)
# The header generated by generate_parameter_library uses a deprecated alias of parameter_traits
target_compile_options(${PROJECT_NAME}_lib PRIVATE -Wno-deprecated-declarations)

rclcpp_components_register_node(${PROJECT_NAME}_lib
  PLUGIN "autoware::safety_planner::SafetyPlannerNode"
  EXECUTABLE ${PROJECT_NAME}_node
)

if(BUILD_TESTING)
  find_package(ament_cmake_ros REQUIRED)

  file(GLOB UNIT_TEST_SOURCES test/unit/*.cpp)
  if(UNIT_TEST_SOURCES)
    ament_add_ros_isolated_gtest(test_${PROJECT_NAME}
      ${UNIT_TEST_SOURCES}
      TIMEOUT 120
    )
    target_link_libraries(test_${PROJECT_NAME}
      ${PROJECT_NAME}_lib
      ${PROJECT_NAME}_common
      ${PROJECT_NAME}_trajectory_planner_plugins
      safety_planner_parameters
    )
    target_include_directories(test_${PROJECT_NAME} PRIVATE
      ${CMAKE_CURRENT_SOURCE_DIR}/src
      ${CMAKE_CURRENT_SOURCE_DIR}/test
    )
    # parameter_traits::OK in the generated parameter header is deprecated
    target_compile_options(test_${PROJECT_NAME} PRIVATE -Wno-deprecated-declarations)

    # Plots of the unit tests (pyplot), written to test_results/unit/
    option(SAFETY_PLANNER_UNIT_TEST_PLOT "Plot the unit test cases with pyplot" OFF)
    if(SAFETY_PLANNER_UNIT_TEST_PLOT)
      message(STATUS "SAFETY_PLANNER_UNIT_TEST_PLOT is ON")
      find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
      find_package(pybind11_vendor REQUIRED)
      find_package(pybind11 REQUIRED)
      target_sources(test_${PROJECT_NAME} PRIVATE
        test/test_output_utils.cpp
        test/test_plot_utils.cpp
      )
      target_link_libraries(test_${PROJECT_NAME} Python3::Python pybind11::embed)
      ament_target_dependencies(test_${PROJECT_NAME} autoware_pyplot)
      target_compile_definitions(test_${PROJECT_NAME} PRIVATE
        SAFETY_PLANNER_UNIT_TEST_PLOT
        TEST_RESULTS_DIR="${CMAKE_CURRENT_BINARY_DIR}/test_results"
      )
    endif()
  endif()

  # Closed-loop simulation (test/closed_loop) that drives SafetyPlanner::plan without a node.
  option(SAFETY_PLANNER_CLOSED_LOOP_TEST "Build and run the closed-loop simulation tests" OFF)
  if(SAFETY_PLANNER_CLOSED_LOOP_TEST)
    message(STATUS "SAFETY_PLANNER_CLOSED_LOOP_TEST is ON")
    find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
    find_package(pybind11_vendor REQUIRED)
    find_package(pybind11 REQUIRED)

    file(GLOB CLOSED_LOOP_TEST_SOURCES test/closed_loop/*.cpp test/*.cpp)
    ament_add_ros_isolated_gtest(test_${PROJECT_NAME}_closed_loop
      ${CLOSED_LOOP_TEST_SOURCES}
      TIMEOUT 600
    )
    target_link_libraries(test_${PROJECT_NAME}_closed_loop
      ${PROJECT_NAME}_lib
      ${PROJECT_NAME}_common
      safety_planner_parameters
      Python3::Python
      pybind11::embed
    )
    target_include_directories(test_${PROJECT_NAME}_closed_loop PRIVATE
      ${CMAKE_CURRENT_SOURCE_DIR}/src
      ${CMAKE_CURRENT_SOURCE_DIR}/test
    )
    ament_target_dependencies(test_${PROJECT_NAME}_closed_loop autoware_pyplot autoware_test_utils)
    target_compile_options(test_${PROJECT_NAME}_closed_loop PRIVATE -Wno-deprecated-declarations)
    target_compile_definitions(test_${PROJECT_NAME}_closed_loop PRIVATE
      TEST_RESULTS_DIR="${CMAKE_CURRENT_BINARY_DIR}/test_results"
    )
  endif()
endif()

install(PROGRAMS
  scripts/generate_test_data.py
  DESTINATION lib/${PROJECT_NAME}
)

ament_auto_package(INSTALL_TO_SHARE
  config
  launch
  test_data
)
